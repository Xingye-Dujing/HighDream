=== i=11 ===
\begin{align}
& \left[\begin{matrix}1 & 2\\4 & 2\end{matrix}\right] \quad & \text{输入矩阵 $A$}\\
& \text{} \quad & \text{}\\
& \text{$法\,1:\,特殊结构检测$} \quad & \text{}\\
& \text{} \quad & \text{}\\
& \text{$\Rightarrow 无特殊结构$} \quad & \text{}\\
& \text{} \quad & \text{}\\
& \text{$法\,2:\;$拉普拉斯展开法} \quad & \text{}\\
& \text{} \quad & \text{}\\
&\Rightarrow -6 \quad & \text{$2×2\,行列式$}\\
& \text{} \quad & \text{}\\
& \text{$法\,3:\;$LU/PLU$\,$分解法} \quad & \text{}\\
& \text{} \quad & \text{}\\
&\Rightarrow \left[\begin{matrix}1 & 0\\4 & 1\end{matrix}\right] \quad & \text{$L(下三角矩阵)$}\\
& \text{} \quad & \text{}\\
&\Rightarrow \left[\begin{matrix}1 & 2\\0 & -6\end{matrix}\right] \quad & \text{$U(上三角矩阵)$}\\
& \text{} \quad & \text{}\\
&\Rightarrow -6 \quad & \text{$|A|=|L| \cdot |U|=1 \times -6=-6$}\\
& \text{} \quad & \text{}\\
& \text{$法\,4:\,行变换法(高斯消元法)$} \quad & \text{$假设所有符号表达式不为\,0$}\\
& \text{} \quad & \text{}\\
& \text{在计算行列式时, 即使分类讨论最后也会统一成该假设下的结果} \quad & \text{$所以此处并不需要进行是否为\,0\,的讨论$}\\
& \text{} \quad & \text{}\\
&\Rightarrow \left[\begin{matrix}1 & 2\\0 & -6\end{matrix}\right] \quad & \text{$R_{2} - \left(4\right) \cdot R_{1}\\to R_{2}$}\\
& \text{} \quad & \text{}\\
&\Rightarrow -6 \quad & \text{$已化为上三角矩阵:\;|A|\,=\,对角线元素乘积$}\end{align}
